  <script>
    const viewer = document.getElementById('lureViewer');
    const variantContainer = document.getElementById('variantContainer');
    const modelButtons = document.querySelectorAll('.model-btn');

    function updateVariantButtons() {
      variantContainer.innerHTML = '';
      const variants = viewer.availableVariants;
      if (!variants || variants.length === 0) return;

      variants.forEach((vName, index) => {
        const btn = document.createElement('button');
        btn.className = `variant-btn ${index === 0 ? 'active' : ''}`;
        btn.textContent = vName;

        btn.addEventListener('click', () => {
          viewer.variantName = vName;
          document.querySelectorAll('.variant-btn').forEach(b => b.classList.remove('active'));
          btn.classList.add('active');
        });

        variantContainer.appendChild(btn);
      });
    }

    viewer.addEventListener('load', updateVariantButtons);

    modelButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        if (btn.disabled) return;
        modelButtons.forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        viewer.src = btn.getAttribute('data-src');
      });
    });
  </script>
